cmake_minimum_required(VERSION 3.18)
project(noahmp_ccpp LANGUAGES Fortran)
find_package(Python3 REQUIRED COMPONENTS Interpreter)

set(CCPP_PHYSICS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../ccpp-scm/ccpp/physics"
    CACHE PATH "CCPP-Physics checkout with the modular Machine compatibility constants")
get_filename_component(CCPP_SCM_ROOT "${CCPP_PHYSICS_DIR}/../.." ABSOLUTE)
list(APPEND CMAKE_MODULE_PATH "${CCPP_SCM_ROOT}/CMakeModules/Modules")
find_package(NetCDF REQUIRED COMPONENTS Fortran)

option(SINGLE_PREC "Use CCPP single precision" OFF)
set(machine_source "${CCPP_PHYSICS_DIR}/physics/hooks/machine.F")
if(NOT EXISTS "${machine_source}")
  message(FATAL_ERROR "Set CCPP_PHYSICS_DIR to the CCPP-Physics checkout")
endif()
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/ccpp_sources.py
                OUTPUT_VARIABLE noahmp_sources OUTPUT_STRIP_TRAILING_WHITESPACE
                RESULT_VARIABLE manifest_status)
if(NOT manifest_status EQUAL 0)
  message(FATAL_ERROR "Cannot resolve modular Noah-MP source dependencies")
endif()
add_library(noahmp_ccpp STATIC ${machine_source} ${noahmp_sources})
target_link_libraries(noahmp_ccpp PUBLIC NetCDF::NetCDF_Fortran)
set_target_properties(noahmp_ccpp PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/modules")
target_include_directories(noahmp_ccpp PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/modules")
target_compile_definitions(noahmp_ccpp PRIVATE CCPP)
if(SINGLE_PREC)
  target_compile_definitions(noahmp_ccpp PRIVATE SINGLE_PREC)
endif()
if(CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
  target_compile_options(noahmp_ccpp PRIVATE -cpp -ffree-line-length-none)
  if(NOT SINGLE_PREC)
    target_compile_options(noahmp_ccpp PRIVATE -fdefault-real-8 -fdefault-double-8)
  endif()
endif()
